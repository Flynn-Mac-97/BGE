### the beacon module as the dev server transforms it
import { createHotContext as __vite__createHotContext } from "/@vite/client";import.meta.hot = __vite__createHotContext("/@id/__x00__virtual:engine-tab-beacon");
const hot = import.meta.hot
if (hot) {
  const key = 'engine:tab-id'
  let id = sessionStorage.getItem(key)
  if (!id) { id = Math.random().toString(36).slice(2, 10); sessionStorage.setItem(key, id) }
  const announce = () => hot.send('engine:tab', {
    id,
    url: location.href,
    title: document.title,
    hidden: document.hidden
  })
  announce()
  addEventListener('visibilitychange', announce)
}
